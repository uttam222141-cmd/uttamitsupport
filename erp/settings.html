// Include this on any doc page (invoice, quotation, job card, challan, amc, // receipt) AFTER auth-guard.js. It fills [data-cf="..."] elements from the // company_settings table, and — only for brand-new (unsaved) documents — // prefills the default GST % and tax type. document.addEventListener('erp:ready', async () => { const { data, error } = await window.supabaseClient .from('company_settings') .select('*') .eq('id', 1) .single(); if (error || !data) return; window.companySettings = data; document.querySelectorAll('[data-cf]').forEach((el) => { const field = el.getAttribute('data-cf'); const value = data[field]; if (value == null) return; if (el.tagName === 'IMG') { el.src = value; } else { el.textContent = value; } }); // Only prefill defaults on a fresh, unsaved document — never overwrite a // document already loaded from the database for editing. const isExistingDoc = new URLSearchParams(window.location.search).get('id'); if (isExistingDoc) return; const gstField = document.getElementById('t-gst-percent'); if (gstField && data.default_gst_percent != null) { gstField.value = data.default_gst_percent; } const taxTypeField = document.querySelector(`input[name="tax-type"][value="${data.default_tax_type}"]`); if (taxTypeField) { taxTypeField.checked = true; taxTypeField.dispatchEvent(new Event('change')); } });
